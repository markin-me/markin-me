function paymentIconClass(code) {
    const c = String(code || "").toLowerCase();
    if (c.includes("cash")) return "fa-money-bill-wave";
    if (c.includes("sbp")) return "fa-qrcode";
    if (c.includes("card")) return "fa-credit-card";
    return "fa-credit-card";
  }

  function formatDateTimeNumeric(value) {
    if (!value) return "—";
    const d = new Date(String(value).replace(" ", "T"));
    if (Number.isNaN(d.getTime())) return String(value);
    const dd = String(d.getDate()).padStart(2, "0");
    const mm = String(d.getMonth() + 1).padStart(2, "0");
    const yyyy = String(d.getFullYear());
    const hh = String(d.getHours()).padStart(2, "0");
    const mi = String(d.getMinutes()).padStart(2, "0");
    return `${dd}.${mm}.${yyyy}, ${hh}:${mi}`;
  }

  function parseOrderItems(order) {
    if (!order) return [];
    if (Array.isArray(order.items)) return order.items;
    if (typeof order.items === "string") {
      try {
        const parsed = JSON.parse(order.items);
        return Array.isArray(parsed) ? parsed : [];
      } catch {
        return [];
      }
    }
    return [];
  }

  function getOrderItemLineTotal(item) {
    const qty = Math.max(1, Number(item?.qty || item?.quantity || 1));
    const direct = Number(item?.line_total ?? item?.total ?? item?.total_price ?? Number.NaN);
    if (Number.isFinite(direct)) return direct;
    return Number(item?.price || 0) * qty;
  }

  function renderOrderItemsHtml(items) {
    const list = Array.isArray(items) ? items : [];
    if (!list.length) return '<div class="muted">—</div>';

    return list.map((item, idx) => {
      const name = escapeHtml(item?.product_name || item?.name || item?.combo_title || "Товар");
      const qty = Math.max(1, Number(item?.qty || item?.quantity || 1));
      const lineTotal = getOrderItemLineTotal(item);
      const oldLineTotal = Number(item?.old_line_total || item?.discount?.original_line_total || 0);
      const showOld = oldLineTotal > lineTotal;
      const photos = Array.isArray(item?.photos) ? item.photos.filter(Boolean) : [];
      const photoHtml = photos.length
        ? `<div class="order-item-photo-small"><img src="${escapeHtml(photos[0])}" alt="${name}" /></div>`
        : "";

      const detailLines = [];
      const variants = Array.isArray(item?.variants) ? item.variants : [];
      variants.forEach((v) => {
        const line = [v?.label || v?.value || "", v?.group_title || ""].join(" ").trim();
        if (line) detailLines.push(`<div class="order-item-composition-item">&bull; ${escapeHtml(line)}</div>`);
      });
      const ingredients = Array.isArray(item?.ingredients) ? item.ingredients : [];
      ingredients.forEach((ing) => {
        const qtyNum = Number(ing?.quantity ?? ing?.qty ?? 0);
        if (!Number.isFinite(qtyNum) || qtyNum <= 0) return;
        const unit = String(ing?.unit_label || ing?.unit || "").trim();
        const nameLine = String(ing?.ingredient_name || ing?.name || "").trim();
        const line = `${qtyNum}${unit ? unit : ""} ${nameLine}`.trim();
        if (line) detailLines.push(`<div class="order-item-composition-item">&bull; ${escapeHtml(line)}</div>`);
      });
      const options = Array.isArray(item?.options) ? item.options : [];
      options.forEach((opt) => {
        const qtyNum = Number(opt?.qty || opt?.quantity || 0);
        if (!Number.isFinite(qtyNum) || qtyNum <= 0) return;
        const variant = String(opt?.variant_label || "").trim();
        const title = String(opt?.title || opt?.name || "").trim();
        const line = variant ? `${variant} ${title}`.trim() : `${qtyNum}шт ${title}`.trim();
        if (line) detailLines.push(`<div class="order-item-composition-item">&bull; ${escapeHtml(line)}</div>`);
      });

      const detailsHtml = detailLines.length
        ? `<div class="order-item-composition">${detailLines.join("")}</div>`
        : "";
      const titleHtml = `${name} x ${qty}`;
      const priceHtml = showOld
        ? `<span class="order-item-old-price">${money(oldLineTotal)}</span><span class="order-item-price-current">${money(lineTotal)}</span>`
        : `<span class="order-item-price-current">${money(lineTotal)}</span>`;

      return `
        <div class="order-item" data-item-idx="${idx}">
          <div class="order-item-line">
            ${photoHtml}
            <div class="order-item-content">
              <div class="order-item-title">${titleHtml}</div>
              ${detailsHtml}
              <div class="order-item-footer">
                <div class="order-item-price">${priceHtml}</div>
              </div>
            </div>
          </div>
        </div>
      `;
    }).join("");
  }

  function computeOrderSummary(order, items) {
    const itemsSubtotal = (Array.isArray(items) ? items : [])
      .reduce((sum, item) => sum + getOrderItemLineTotal(item), 0);
    const subtotalRaw = Number(order?.subtotal ?? order?.items_total ?? order?.items_subtotal ?? itemsSubtotal);
    const subtotal = Number.isFinite(subtotalRaw) ? subtotalRaw : itemsSubtotal;

    let discount = Number(order?.discount_total ?? order?.total_discount ?? 0);
    if (!(discount > 0)) {
      const total = Number(order?.total_price || 0);
      const delivery = Number(order?.delivery_cost || 0);
      const candidate = subtotal + delivery - total;
      if (candidate > 0.01) discount = candidate;
    }

    return {
      subtotal,
      discount: discount > 0 ? discount : 0,
    };
  }

  function showClientOrderInfoEmpty(title, text) {
    if (!orderInfoEls) return;
    setNodesHidden(orderInfoEls.empty, false);
    setNodesHidden(orderInfoEls.content, true);
    if (title) setNodesText(orderInfoEls.emptyTitle, title);
    if (text) setNodesText(orderInfoEls.emptyText, text);
  }

  function renderClientOrderInfo(order) {
    if (!orderInfoEls || !order) return;

    const items = parseOrderItems(order);
    const summary = computeOrderSummary(order, items);

    setNodesHidden(orderInfoEls.empty, true);
    setNodesHidden(orderInfoEls.content, false);

    setNodesText(orderInfoEls.title, `ЗАКАЗ #${order.id}`);
    setNodesText(orderInfoEls.meta, formatDateTimeNumeric(order.created_at));
    setNodesText(orderInfoEls.status, String(order.status_title || "—"));
    setNodesText(orderInfoEls.clientName, String(order.customer_name || "—"));

    const phone = String(order.customer_phone || "").trim();
    const phoneDisplay = phone || "—";
    setNodesText(orderInfoEls.clientPhone, phoneDisplay);
    const clientId = Number(order.customer_id || 0);
    const canOpenClient = (Number.isFinite(clientId) && clientId > 0) || !!phone;
    orderInfoEls.clientPhone.forEach((el) => {
      if (!el) return;
      if (canOpenClient) {
        el.setAttribute("href", "#");
        el.setAttribute("data-action", "open-client");
        if (clientId > 0) el.setAttribute("data-client-id", String(clientId));
        else el.removeAttribute("data-client-id");
        if (phone) el.setAttribute("data-client-phone", phone);
        else el.removeAttribute("data-client-phone");
      } else {
        el.removeAttribute("href");
        el.removeAttribute("data-action");
        el.removeAttribute("data-client-id");
        el.removeAttribute("data-client-phone");
      }
    });

    const interval = String(order.time_option_title || "").trim();
    setNodesText(orderInfoEls.deliveryInterval, interval || "—");
    setNodesHidden(orderInfoEls.deliveryIntervalRow, !interval);
    const urgent = Boolean(order.is_urgent || order.urgent || order.time_option_code === "urgent");
    setNodesHidden(orderInfoEls.urgent, !urgent);

    const isPickup = String(order.method_code || "") === "pickup";
    setNodesText(orderInfoEls.deliveryAddressTitle, isPickup ? "Адрес самовывоза" : "Адрес доставки");

    let address = String(order.address || "").trim();
    if (!address && order.pickup_store_address) {
      address = order.pickup_store_name
        ? `${order.pickup_store_name}, ${order.pickup_store_address}`
        : String(order.pickup_store_address || "");
    }
    setNodesText(orderInfoEls.deliveryAddress, address || "—");

    const addressComment = String(order.address_comment || "").trim();
    setNodesText(orderInfoEls.deliveryAddressCommentText, addressComment);
    setNodesHidden(orderInfoEls.deliveryAddressComment, !addressComment);

    const orderComment = String(order.comment || "").trim();
    setNodesText(orderInfoEls.orderCommentText, orderComment);
    setNodesHidden(orderInfoEls.orderCommentBlock, !orderComment);

    setNodesHtml(orderInfoEls.itemsList, renderOrderItemsHtml(items));

    setNodesText(orderInfoEls.payMethod, String(order.payment_title || "—"));
    const paymentIcon = paymentIconClass(order.payment_code);
    orderInfoEls.payIcon.forEach((el) => {
      if (!el) return;
      el.innerHTML = `<i class="fas ${paymentIcon}"></i>`;
    });

    const total = Number(order.total_price || 0);
    const deliveryCost = Number(order.delivery_cost || 0);
    setNodesText(orderInfoEls.deliveryCost, money(deliveryCost));
    setNodesHidden(orderInfoEls.deliveryRow, false);

    const changeFrom = Number(order.change_from || 0);
    const changeAmount = changeFrom > total ? (changeFrom - total) : 0;
    setNodesText(orderInfoEls.changeFrom, money(changeFrom));
    setNodesText(orderInfoEls.changeAmount, money(changeAmount));
    setNodesHidden(orderInfoEls.changeFromRow, !(changeFrom > 0));
    setNodesHidden(orderInfoEls.changeAmountRow, !(changeFrom > 0));

    setNodesText(orderInfoEls.subtotal, money(summary.subtotal));
    setNodesHidden(orderInfoEls.subtotalRow, !(summary.discount > 0));
    setNodesText(orderInfoEls.discountAmount, `-${money(summary.discount)}`);
    setNodesHidden(orderInfoEls.discountRow, !(summary.discount > 0));
    setNodesHidden(orderInfoEls.discountInfoBtn, true);
    setNodesHidden(orderInfoEls.discountBreakdown, true);

    setNodesText(orderInfoEls.total, money(total));

    setNodesHidden(orderInfoEls.statusMenu, true);
    orderInfoEls.statusToggle.forEach((btn) => {
      if (!btn) return;
      btn.setAttribute("aria-expanded", "false");
      btn.setAttribute("disabled", "disabled");
    });
  }

  async function activateOrderById(orderId) {
    const id = Number(orderId || 0);
    if (!Number.isFinite(id) || id <= 0) return;

    state.activeOrderId = id;
    state.activeOrder = null;

    const cached = state.orderCache.get(id);
    if (cached) {
      state.activeOrder = cached;
      renderClientOrderInfo(cached);
      return;
    }

    showClientOrderInfoEmpty("Загрузка заказа", "Подождите...");

    try {
      const json = await apiJson(`/api/admin/orders/${id}`);
      const order = json?.data || null;
      if (!order) {
        showClientOrderInfoEmpty("Ошибка", "Не удалось загрузить заказ");
        return;
      }
      state.activeOrder = order;
      state.orderCache.set(id, order);
      renderClientOrderInfo(order);
    } catch (err) {
      console.error(err);
      showClientOrderInfoEmpty("Ошибка", "Не удалось загрузить заказ");
    }
  }

  function openOrderTab(orderId) {
    const id = Number(orderId || 0);
    if (!Number.isFinite(id) || id <= 0) return;

    ensureTab({
      type: "order",
      id,
      title: `№${id}`,
      onActivate: () => activateOrderById(id),
    });

    if (isMobile()) openSheet();
  }
